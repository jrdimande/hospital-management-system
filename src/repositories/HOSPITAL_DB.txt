CREATE TABLE patients(

	id INT PRIMARY KEY AUTO_INCREMENT,
	name VARCHAR(100),
	age INT,
	gender ENUM('MALE', 'FEMALE'),
	phone VARCHAR(11),
	address VARCHAR(100)


) 


CREATE TABLE appointments(

	id INT PRIMARY KEY AUTO_INCREMENT,
	patient VARCHAR(100),
	doctor VARCHAR(100),
	date VARCHAR(30),
	notes VARCHAR(350)
)

CREATE TABLE doctors(

	id INT PRIMARY KEY AUTO_INCREMENT,
	name VARCHAR(100),
	speciality VARCHAR(100),
	phone VARCHAR(11),
	password VARCHAR(8)


);
